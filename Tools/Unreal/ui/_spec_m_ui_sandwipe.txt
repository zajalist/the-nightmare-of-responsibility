"""M_UI_SandWipe: full-screen black-sand transition (splash cards, wake-up, day rollover). Run via ue_remote.py.
Coverage 0 = clear, 1 = screen buried in sand. Sand falls in from the top: the dissolve noise is biased by the
row so the edge travels downward, using the same MF_UI_SandFlow / MF_UI_SandDissolve as the sand button.
Wipe = (Coverage, EdgeWidth, FallBias, Seed)."""
exec(open('P:/FoA/Tools/Unreal/ui/mat_graph.py').read())
AT = unreal.AssetToolsHelpers.get_asset_tools()
EAL = unreal.EditorAssetLibrary
PATH = '/Game/FoA/UI/Materials/M_UI_SandWipe'
mpc = unreal.load_asset('/Game/FoA/UI/Style/MPC_UI')
m = unreal.load_asset(PATH) or AT.create_asset('M_UI_SandWipe', '/Game/FoA/UI/Materials', unreal.Material,
                                                unreal.MaterialFactoryNew())
m.set_editor_property('material_domain', unreal.MaterialDomain.MD_UI)
m.set_editor_property('blend_mode', unreal.BlendMode.BLEND_TRANSLUCENT)
g = MatGraph(PATH).clear()
V = unreal.MaterialExpressionVectorParameter
uv = g.add('MaterialExpressionTextureCoordinate', -1280, 0)
wipe = g.param(V, 'Wipe', -1280, 192, 'Wipe', unreal.LinearColor(0.0, 0.12, 0.55, 0.0), 0,
               'R Coverage 0-1, G EdgeWidth, B FallBias (0 pure noise, 1 straight curtain), A Seed')
tiling = g.add('MaterialExpressionConstant2Vector', -1280, -160, r=7.5, g=4.22, desc='1920x1080 / 256')
zero = g.add('MaterialExpressionConstant', -1280, 416, r=0.0, desc='static: no flow')
flow = g.call('/Game/FoA/UI/Materials/Functions/MF_UI_SandFlow', -992, 0)
row = g.add('MaterialExpressionComponentMask', -992, 256, r=False, g=True, b=False, a=False)
fall = g.add('MaterialExpressionLinearInterpolate', -736, 96)          # lerp(noise, V, FallBias)
diss = g.call('/Game/FoA/UI/Materials/Functions/MF_UI_SandDissolve', -480, 96)
cover = g.add('MaterialExpressionOneMinus', -224, 32)                  # sand where the dissolve says "gone"
soot = g.add('MaterialExpressionCollectionParameter', -480, -160, collection=mpc, parameter_name='Soot')
ink = g.add('MaterialExpressionCollectionParameter', -480, -288, collection=mpc, parameter_name='Ink')
glint = g.add('MaterialExpressionMultiply', -224, -224, const_b=0.35)
color = g.add('MaterialExpressionLinearInterpolate', 32, -128)
g.link(uv, '', flow, 'UV'); g.link(tiling, '', flow, 'Tiling'); g.link(wipe, 'A', flow, 'Seed'); g.link(zero, '', flow, 'Phase')
g.link(uv, '', row, '')
g.link(flow, 'Dissolve', fall, 'A'); g.link(row, '', fall, 'B'); g.link(wipe, 'B', fall, 'Alpha')
g.link(fall, '', diss, 'Noise'); g.link(wipe, 'R', diss, 'Amount'); g.link(wipe, 'G', diss, 'Edge')
g.link(diss, 'Opacity', cover, '')
g.link(ink, '', glint, 'A')
g.link(soot, '', color, 'A'); g.link(glint, '', color, 'B'); g.link(diss, 'Edge', color, 'Alpha')
g.out(color, '', unreal.MaterialProperty.MP_EMISSIVE_COLOR); g.out(cover, '', unreal.MaterialProperty.MP_OPACITY)
g.comment('Inputs', 'One packed vector; tiling is fixed for a 1080p-tall screen (1 texel per px)', -1312, -224, 256, 720, 'input')
g.comment('Falling sand noise', 'Static noise lerped toward the row so the sand edge travels top to bottom',
          -1024, -64, 512, 480, 'noise')
g.comment('Bury the frame', 'Inverted shared dissolve: Coverage 1 = solid black sand, glints on the moving edge',
          -512, -352, 672, 576, 'fx')
g.finish()
p = '/Game/FoA/UI/Materials/MI_UI_SandWipe'
inst = unreal.load_asset(p) or AT.create_asset('MI_UI_SandWipe', '/Game/FoA/UI/Materials', unreal.MaterialInstanceConstant,
                                               unreal.MaterialInstanceConstantFactoryNew())
MEL.set_material_instance_parent(inst, m)
MEL.update_material_instance(inst)
print('MI_UI_SandWipe', EAL.save_asset(p, only_if_is_dirty=False))
